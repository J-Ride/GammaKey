**Tech Stack:** C# .NET 8 WinForms, x64, xUnit. Spec: `docs/superpowers/specs/2026-11-09-v1-fixes-design.md`.

```csharp
    public AppSettings Load()
    {
        if (!File.Exists(Path)) return Defaults();

        var settings = TryParse(File.ReadAllText(Path));
        if (settings is null)
        {
            // A half-written or hand-edited file must never lose the user's profiles silently,
            // so keep a copy next to it before falling back.
            File.Copy(Path, Path + ".corrupt", overwrite: true);
            return Defaults();
        }
        return settings;
    }

    // Shared by Load and by profile import, so an imported file gets the same clamping and
    // null fixes as the app's own file.
    public static AppSettings? TryParse(string json)
    {
        AppSettings? settings;
        try
        {
            settings = JsonSerializer.Deserialize<AppSettings>(json, JsonOptions);
        }
        catch (JsonException)
        {
            return null;
        }
        if (settings is null) return null;

        settings.Profiles ??= new List<Profile>();
        settings.Profiles.RemoveAll(p => p is null);

        foreach (var profile in settings.Profiles)
        {
            profile.Hotkey ??= "";
            profile.Name ??= "";
            profile.Values = profile.Values.Clamped();
            profile.Target ??= MonitorTarget.ForPrimary();
            if (string.IsNullOrWhiteSpace(profile.Id)) profile.Id = Guid.NewGuid().ToString("N");
        }
        settings.ResetHotkey ??= "";
        return settings;
    }
```

```csharp
using GammaKey.Models;

namespace GammaKey.Storage;

public static class ProfileImport
{
    public sealed record Result(IReadOnlyList<Profile> Added, int HotkeysCleared);

    // Import only ever adds. A friend's file must never replace a tuned profile, and two
    // profiles must never share a hotkey, so clashes get a suffix or a cleared key.
    public static Result Merge(IReadOnlyList<Profile> existing, IReadOnlyList<Profile> incoming, string resetHotkey)
    {
        var names = new HashSet<string>(existing.Select(p => p.Name), StringComparer.OrdinalIgnoreCase);
        var hotkeys = new HashSet<string>(existing.Select(p => p.Hotkey).Where(h => h.Length > 0), StringComparer.OrdinalIgnoreCase);
        if (resetHotkey.Length > 0) hotkeys.Add(resetHotkey);

        var added = new List<Profile>(incoming.Count);
        int cleared = 0;
        foreach (var source in incoming)
        {
            var profile = source.Clone();
            profile.Hotkey = source.Hotkey;
            profile.Name = UniqueName(source.Name, names);
            names.Add(profile.Name);
            if (profile.Hotkey.Length > 0 && !hotkeys.Add(profile.Hotkey))
            {
                profile.Hotkey = "";
                cleared++;
            }
            added.Add(profile);
        }
        return new Result(added, cleared);
    }

    private static string UniqueName(string name, HashSet<string> taken)
    {
        if (!taken.Contains(name)) return name;
        var candidate = name + " (imported)";
        for (int n = 2; taken.Contains(candidate); n++) candidate = $"{name} (imported {n})";
        return candidate;
    }
}
```

```csharp
    private void ExportProfiles()
    {
        using var dialog = new SaveFileDialog { Filter = ProfileFileFilter, FileName = "GammaKey-profiles.json", DefaultExt = "json" };
        if (dialog.ShowDialog(this) != DialogResult.OK) return;
        try
        {
            // Same shape as profiles.json, so an export and a copied profiles.json both import.
            File.WriteAllText(dialog.FileName, JsonSerializer.Serialize(_controller.Settings, ProfileStore.JsonOptions));
            _status.Text = $"Exported {_controller.Settings.Profiles.Count} profiles.";
        }
        catch (Exception e) when (e is IOException or UnauthorizedAccessException)
        {
            _status.Text = "Could not write that file.";
        }
    }

    private void ImportProfiles()
    {
        using var dialog = new OpenFileDialog { Filter = ProfileFileFilter };
        if (dialog.ShowDialog(this) != DialogResult.OK) return;

        AppSettings? incoming = null;
        try
        {
            incoming = ProfileStore.TryParse(File.ReadAllText(dialog.FileName));
        }
        catch (Exception e) when (e is IOException or UnauthorizedAccessException)
        {
        }
        if (incoming is null || incoming.Profiles.Count == 0)
        {
            _status.Text = "Could not read that file. Is it a GammaKey profiles file?";
            return;
        }

        var result = ProfileImport.Merge(_controller.Settings.Profiles, incoming.Profiles, _controller.Settings.ResetHotkey);
        int firstNew = _controller.Settings.Profiles.Count;
        _controller.Settings.Profiles.AddRange(result.Added);
        SaveSettingsAndRegister();
        _profileList.SelectedIndex = firstNew;
        _status.Text = result.HotkeysCleared switch
        {
            0 => $"Imported {result.Added.Count} profiles.",
            1 => $"Imported {result.Added.Count} profiles. 1 hotkey cleared because it was in use.",
            var k => $"Imported {result.Added.Count} profiles. {k} hotkeys cleared because they were in use.",
        };
    }
```
